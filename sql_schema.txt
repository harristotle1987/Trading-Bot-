-- PostgreSQL Schema for Supabase/Neon

CREATE TABLE trades (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    account_mode VARCHAR(10) NOT NULL CHECK (account_mode IN ('DEMO', 'LIVE')),
    broker VARCHAR(50) NOT NULL,
    symbol VARCHAR(20) NOT NULL,
    side VARCHAR(10) NOT NULL CHECK (side IN ('BUY', 'SELL')),
    quantity DECIMAL(18, 8) NOT NULL,
    entry_price DECIMAL(18, 8) NOT NULL,
    current_mark_price DECIMAL(18, 8),
    stop_loss DECIMAL(18, 8),
    take_profit DECIMAL(18, 8),
    unrealized_pnl DECIMAL(18, 2) DEFAULT 0.00,
    ai_confidence_score DECIMAL(5, 2),
    status VARCHAR(20) NOT NULL DEFAULT 'OPEN',
    timeframe VARCHAR(10),
    reasoning TEXT,
    opened_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    closed_at TIMESTAMP WITH TIME ZONE
);

CREATE TABLE system_balances (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    mode VARCHAR(10) NOT NULL UNIQUE CHECK (mode IN ('DEMO', 'LIVE')),
    balance DECIMAL(18, 2) NOT NULL DEFAULT 0.00,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);
